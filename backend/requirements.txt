# Backend runtime dependencies.
#
# Keep this file ASCII-only: pip decodes requirements files using the locale
# encoding on Windows, and non-ASCII comments make installation fail outright
# with a UnicodeDecodeError before anything is installed.
#
# Only packages that are imported directly are listed. The following are
# deliberately kept even though nothing imports them by name:
#   - langchain-core: the API surface of langchain-openai, pinned so an upstream
#     breaking change cannot slip in
#   - python-multipart: required by FastAPI/Starlette to parse forms
#
# Versions are pinned exactly, to the versions the test suite was run against.
# Loose ranges mean a fresh `pip install` can pull a new major with breaking
# changes and nobody finds out until deployment; the frontend pins through
# package-lock.json for the same reason.
#
# Everything else arrives transitively and is intentionally not repeated here:
#   annotated-types (pydantic), tenacity (langchain-core),
#   pandas + numpy + beautifulsoup4 (yfinance)

# FastAPI and Server
fastapi==0.141.1
uvicorn[standard]==0.54.0
python-multipart==0.0.32

# Database
sqlalchemy==2.1.1
pymysql==1.2.3

# LLM (all clients are built by app/services/llm_provider.py)
langchain-core==1.6.5
langchain-openai==1.6.6
openai==3.20.0

# Validation
pydantic==2.13.5
pydantic-settings==2.15.0

# External Data
yfinance==1.7.0
requests==2.34.2
feedparser==6.0.14
# GPR official daily index is a legacy .xls workbook; xlrd 2.x reads exactly
# that format (and nothing else), so it stays pinned alongside the other
# external-data clients.
xlrd==2.0.2

# Scheduler
apscheduler==3.11.3
# zoneinfo needs a timezone database: slim images ship without system tzdata,
# and without it Asia/Shanghai cannot be loaded. Locally it arrives via
# pandas/tzlocal, but the production container installs only this file, so it
# has to be listed explicitly.
tzdata==2026.4

# Utilities
python-dotenv==1.2.3
loguru==0.7.3
# [socks] pulls in socksio. If the host sets a SOCKS proxy, httpx raises while
# building a client without it, which makes every LLM call fail (see LLM_TRUST_ENV).
httpx[socks]==0.28.1
